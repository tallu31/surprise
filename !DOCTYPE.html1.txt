// ===== Candles =====
const candleCount = 7;
const candlesArea = document.getElementById('candlesArea');
let litCount = 0;
for (let i = 0; i < candleCount; i++) {
const wrap = document.createElement('div');
wrap.className = 'candle-wrap';
wrap.innerHTML = `
<div style="position:relative">
<div class="candle-glow" id="glow${i}"></div>
<div class="flame" id="flame${i}"></div>
<div class="candle"></div>
</div>
<div class="candle-base"></div>
`;
wrap.addEventListener('click', () => toggleCandle(i));
candlesArea.appendChild(wrap);
}
function toggleCandle(i) {
const flame = document.getElementById('flame' + i);
const glow = document.getElementById('glow' + i);
if (flame.classList.contains('lit')) {
flame.classList.remove('lit');
glow.classList.remove('lit');
litCount = Math.max(0, litCount - 1);
} else {
flame.classList.add('lit');
glow.classList.add('lit');
litCount++;
createSparkle(flame);
}
updateCandleMsg();
}
function lightAllCandles() {
for (let i = 0; i < candleCount; i++) {
const flame = document.getElementById('flame' + i);
const glow = document.getElementById('glow' + i);
if (!flame.classList.contains('lit')) {
setTimeout(() => {
flame.classList.add('lit');
glow.classList.add('lit');
createSparkle(flame);
}, i * 150);
}
}
litCount = candleCount;
setTimeout(updateCandleMsg, candleCount * 150 + 100);
}
function blowOutCandles() {
for (let i = 0; i < candleCount; i++) {
document.getElementById('flame' + i).classList.remove('lit');
document.getElementById('glow' + i).classList.remove('lit');
}
litCount = 0;
updateCandleMsg();
}
function updateCandleMsg() {
const msg = document.getElementById('candleMsg');
if (litCount === 0) msg.textContent = 'Tap the candles to light them ✨';
else if (litCount < candleCount) msg.textContent = `${litCount} of ${candleCount} candles glowing for you 💕`;
else msg.textContent = 'All candles lit — our love shines forever 🔥💖';
}
function createSparkle(el) {
const rect = el.getBoundingClientRect();
for (let i = 0; i < 5; i++) {
const sp = document.createElement('div');
sp.textContent = '✨';
sp.style.cssText = `position:fixed;left:${rect.left + rect.width/2}px;top:${rect.top}px;font-size:12px;pointer-events:none;z-index:999;transition:all 0.8s ease-out;`;
document.body.appendChild(sp);
requestAnimationFrame(() => {
sp.style.transform = `translate(${(Math.random()-0.5)*60}px, ${-30-Math.random()*40}px) scale(0)`;
sp.style.opacity = '0';
});
setTimeout(() => sp.remove(), 900);
}
}

// ===== Confetti =====
function triggerConfetti() {
const colors = ['#fb7185', '#c084fc', '#fbbf24', '#f472b6', '#e879f9', '#fda4af'];
confetti({
particleCount: 120,
spread: 80,
origin: { y: 0.6 },
colors: colors,
disableForReducedMotion: true
});
setTimeout(() => {
confetti({ particleCount: 60, angle: 60, spread: 55, origin: { x: 0, y: 0.7 }, colors });
confetti({ particleCount: 60, angle: 120, spread: 55, origin: { x: 1, y: 0.7 }, colors });
}, 250);
}
// ===== Sprinkles =====
function triggerSprinkles() {
const emojis = ['💕','✨','🌸','💖','⭐','🌹','💗','🎀','💫','🥰','💘','🌺'];
for (let i = 0; i < 35; i++) {
setTimeout(() => {
const s = document.createElement('div');
s.className = 'sprinkle';
s.textContent = emojis[Math.floor(Math.random() * emojis.length)];
s.style.left = Math.random() * 100 + 'vw';
s.style.top = '-20px';
s.style.fontSize = (14 + Math.random() * 16) + 'px';
s.style.animationDuration = (1.8 + Math.random() * 1.5) + 's';
document.body.appendChild(s);
setTimeout(() => s.remove(), 3000);
}, i * 50);
}
}
